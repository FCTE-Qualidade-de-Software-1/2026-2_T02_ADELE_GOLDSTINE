\begin{simbolos}
  \item[$ A $] Contagem de ocorrências favoráveis (ou ausências) usada no numerador de uma métrica
  \item[$ B $] Contagem total de itens verificados, usada no denominador de uma métrica
  \item[$ G_i $] Objetivo de medição GQM ($i = 1, \ldots, 4$)
  \item[$ I $] Nota de impacto de uma característica (1 a 5)
  \item[$ M_j $] Métrica GQM ($j = 1, \ldots, 20$)
  \item[$ P $] Prioridade de uma característica, $P = I \times R$
  \item[$ Q_k $] Questão GQM ($k = 1, \ldots, 14$)
  \item[$ R $] Nota de risco de uma característica (1 a 5)
  \item[$ T_e $] Tempo de execução de uma tarefa por usuário experiente
  \item[$ T_p $] Tempo médio de execução de uma tarefa pelos participantes
  \item[$ X $] Valor calculado de uma métrica
\end{simbolos}
